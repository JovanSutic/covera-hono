### Step 3.1 — Contract

**Touch.** `src/modules/special_access/special_access.contract.ts` (create)

**Do.**

```ts
import { createRoute } from "@hono/zod-openapi";
import {
  AccessIdParamSchema,
  CreateSpecialAccessResponseSchema,
  CreateSpecialAccessSchema,
  RevokeSpecialAccessResponseSchema,
} from "./special_access.schema";
import { authGuard } from "@/middleware/authGuard";
import { rolesGuard } from "@/middleware/roleGuard";
import {
  errorResponses,
  AUTHED,
  ROLE,
  APARTMENT_SCOPED,
  VALIDATION,
} from "@/core/errors/error.responses";

/*
 * The management surface. Both routes are session-only: a token can never mint or
 * revoke a token (D16), which is why neither declares any `SPECIAL_ACCESS_*` code
 * — a request carrying `?accessId=` here is simply ignored, because these routes
 * do not mount `specialAccessGuard`.
 *
 * Ownership is asserted in the handler through the subject's apartment (D16),
 * reusing `assertApartmentOwnership` exactly as `incidents.route.ts` does, so
 * `...APARTMENT_SCOPED` is declared here as it is there.
 */

const createSpecialAccessErrors = errorResponses([
  ...AUTHED,
  ...ROLE,
  ...APARTMENT_SCOPED,
  ...VALIDATION,
  "RESERVATION_NOT_FOUND",
  "INCIDENT_ITEM_NOT_FOUND",
]);

const revokeSpecialAccessErrors = errorResponses([
  ...AUTHED,
  ...ROLE,
  ...APARTMENT_SCOPED,
  ...VALIDATION,
  "SPECIAL_ACCESS_NOT_FOUND",
]);

/**
 * Mints a link. The response carries `accessId` — the only payload that does
 * (D14) — because the host has to relay it; no read route ever returns it.
 *
 * A `host` may mint only for an apartment they own; an `admin` may mint for any.
 */
export const createSpecialAccessRoute = createRoute({
  method: "post",
  path: "/",
  tags: ["Special access"],
  middleware: [authGuard, rolesGuard(["admin", "host"])] as const,
  request: {
    body: {
      content: {
        "application/json": {
          schema: CreateSpecialAccessSchema,
        },
      },
    },
  },
  responses: {
    201: {
      description: "Access link created",
      content: {
        "application/json": {
          schema: CreateSpecialAccessResponseSchema,
        },
      },
    },
    ...createSpecialAccessErrors,
  },
});

/**
 * Cuts a link off. Idempotent: revoking an already-revoked grant is a 200 (D15),
 * so a retried call cannot look like a failure to a client that cannot tell a lost
 * response from a lost grant. An unknown token is a 404.
 */
export const revokeSpecialAccessRoute = createRoute({
  method: "delete",
  path: "/{accessId}",
  tags: ["Special access"],
  middleware: [authGuard, rolesGuard(["admin", "host"])] as const,
  request: {
    params: AccessIdParamSchema,
  },
  responses: {
    200: {
      description: "Access link revoked",
      content: {
        "application/json": {
          schema: RevokeSpecialAccessResponseSchema,
        },
      },
    },
    ...revokeSpecialAccessErrors,
  },
});
```

**Accept.** `npx tsc --noEmit` → exit 0.

**Verify.** `npm run gate`

### Step 3.2 — Route handlers

**Touch.** `src/modules/special_access/special_access.route.ts` (create)

**Do.**

```ts
import { createModuleApp } from "@/core/utils/module-app";
import {
  createSpecialAccessRoute,
  revokeSpecialAccessRoute,
} from "./special_access.contract";
import { specialAccessService } from "./special_access.service";
import { specialAccessErrors } from "./special_access.errors";
import { incidentErrors } from "../incidents/incidents.errors";
import { reservationErrors } from "../reservations/reservations.errors";
import { assertApartmentOwnership } from "@/core/utils/db-validator";

const app = createModuleApp();

app.openapi(createSpecialAccessRoute, async (c) => {
  const db = c.get("db");
  const body = c.req.valid("json");
  const user = c.get("user");

  if (!user) {
    throw incidentErrors.unauthenticated();
  }

  // The subject decides the apartment, and the apartment is what ownership is
  // asserted against — so a client cannot bind a grant to an apartment it does
  // not own (D13, D16). The schema's `superRefine` already guarantees exactly one
  // subject is present for the scope, so the branch below cannot fall through.
  const subject = await specialAccessService.resolveSubjectApartment(
    db,
    body.scope === "inspection_proof"
      ? { reservationId: body.reservationId! }
      : { incidentItemId: body.incidentItemId! },
  );

  if (!subject) {
    // Reported in the vocabulary of the module that owns the subject, so the
    // caller sees RESERVATION_NOT_FOUND / INCIDENT_ITEM_NOT_FOUND rather than a
    // special-access code about something they did not ask for.
    if (body.scope === "inspection_proof") {
      throw reservationErrors.notFound({ id: body.reservationId! });
    }

    throw incidentErrors.itemNotFound({ itemId: body.incidentItemId! });
  }

  await assertApartmentOwnership(db, subject.apartmentId, {
    userId: user.id,
    role: user.role,
    allowAdmin: true,
  });

  const access = await specialAccessService.create(db, body, subject, user.id);

  return c.json({ access }, 201);
});

app.openapi(revokeSpecialAccessRoute, async (c) => {
  const db = c.get("db");
  const { accessId } = c.req.valid("param");
  const user = c.get("user");

  if (!user) {
    throw incidentErrors.unauthenticated();
  }

  // MANAGEMENT lookup, not the redemption one: a revoked or expired grant must
  // still be findable so that revoking it again is the idempotent 200 of D15
  // rather than a 404 a client cannot distinguish from a lost response.
  const grant = await specialAccessService.findByToken(db, accessId);

  if (!grant) {
    throw specialAccessErrors.notFound({ accessId });
  }

  // Ownership, asserted against the grant's own apartment. Without this any host
  // could revoke a link belonging to another host's apartment — the 403 that this
  // contract already declares would never have been produced.
  await assertApartmentOwnership(db, grant.apartmentId, {
    userId: user.id,
    role: user.role,
    allowAdmin: true,
  });

  await specialAccessService.revokeById(db, grant.id);

  return c.json({ success: true as const, accessId }, 200);
});

export default app;
```

**Verify.** `npm run gate`

### Step 3.4 — Test the management routes

**Touch.** `src/modules/special_access/special_access.test.ts` (create)

**Do.** The full file is 290 lines and is reproduced in
[`reference/special_access.test.ts.txt`](./reference/special_access.test.ts.txt). Copy it verbatim
into `src/modules/special_access/special_access.test.ts`.

Its structure, so the executor can confirm the right file landed:

- `POST /special-access` — mints and returns 201 with `accessId`; defaults `expiresInHours` to 12;
  derives the apartment server-side even when the body supplies one; 400 for a missing
  `reservationId`, a missing `incidentItemId`, both subjects at once, and an `expiresInHours` over
  72; 404 in the *owning module's* vocabulary for an unknown reservation and an unknown item; 403
  for a host who does not own the apartment; 403 `FORBIDDEN_ROLE` for a guest; 401 unauthenticated.
- `DELETE /special-access/{accessId}` — 200 on revoke; 200 when already revoked; 404
  `SPECIAL_ACCESS_NOT_FOUND` for an unknown token; 403 when the caller does not own the grant's
  apartment.

It mocks `./special_access.service`, `../users/users.service` and `@/core/utils/db-validator`, and
uses `createTestApp` from `@/core/utils/test-factory` with `authUser` supplied per test — the same
harness `src/modules/assets/assets.test.ts` uses.

**Accept.** `npx vitest run src/modules/special_access/special_access.test.ts` → **16** tests passing, 0 failing; `npm run gate` → **765** tests passing (749 after phase 2, + 16)

## Phase 3 invariants

- [ ] Neither management route declares a `SPECIAL_ACCESS_*` code — a token cannot manage tokens
- [ ] The mint asserts ownership against the **subject's** apartment, never a body value
- [ ] The revoke asserts ownership against the **grant's** apartment
- [ ] An unknown reservation reports `RESERVATION_NOT_FOUND`, not a special-access code
- [ ] Revoking an already-revoked grant is a 200, not a 409 or 404
